cmake_minimum_required(VERSION 3.15)
project(LuauBenchmarks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LUAU_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_CLI ON CACHE BOOL "" FORCE)
add_subdirectory("${LUAU_SOURCE}" luau)
foreach(kind host regex bigint)
  add_executable(luau-${kind} Host.cpp)
  target_link_libraries(luau-${kind} PRIVATE Luau.Compiler Luau.VM Luau.CodeGen)
endforeach()
target_sources(luau-regex PRIVATE Regex.cpp)
target_compile_definitions(luau-regex PRIVATE BENCH_REGEX)
target_include_directories(luau-regex PRIVATE /opt/homebrew/opt/pcre2/include)
target_link_libraries(luau-regex PRIVATE /opt/homebrew/opt/pcre2/lib/libpcre2-8.a)
target_sources(luau-bigint PRIVATE Bigint.cpp)
target_compile_definitions(luau-bigint PRIVATE BENCH_BIGINT)
target_include_directories(luau-bigint PRIVATE /opt/homebrew/opt/gmp/include)
target_link_libraries(luau-bigint PRIVATE /opt/homebrew/opt/gmp/lib/libgmp.a)
add_executable(luau-source-reader ../source-counts-v1/Reader.cpp)
target_link_libraries(luau-source-reader PRIVATE Luau.Ast)
